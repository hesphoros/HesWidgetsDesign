# ============================================================================
# HesWidgetsDesign 库项目配置
# ============================================================================

project(HesWidgetsDesign LANGUAGES CXX)

# ============================================================================
# 源文件收集 - 自动扫描
# ============================================================================

# 自动收集公共头文件
file(GLOB_RECURSE PUBLIC_HEADERS 
    "include/*.h"
)

# 自动收集私有头文件
file(GLOB_RECURSE PRIVATE_HEADERS 
    "private/*.h"
)

# 自动收集开发组件头文件
file(GLOB_RECURSE DEVELOPER_HEADERS 
    "DeveloperComponents/*.h"
)

# 自动收集源文件
file(GLOB_RECURSE SOURCES 
    "src/*.cpp" 
    "src/*.c"
    "private/*.cpp" 
    "private/*.c"
)

# 平台特定源文件
if(WIN32)
    file(GLOB_RECURSE WIN_SOURCES 
        "DeveloperComponents/*.cpp" 
        "DeveloperComponents/*.c"
    )
    list(APPEND SOURCES ${WIN_SOURCES})
endif()

# 所有头文件（用于IDE项目组织）
set(ALL_HEADERS
    ${PUBLIC_HEADERS}
    ${PRIVATE_HEADERS}
    ${DEVELOPER_HEADERS}
)

# ============================================================================
# Visual Studio 项目文件组织
# ============================================================================
source_group("Public Headers" FILES ${PUBLIC_HEADERS})
source_group("Private Headers" FILES ${PRIVATE_HEADERS})
source_group("Developer Components Headers" FILES ${DEVELOPER_HEADERS})
source_group("Source Files" FILES ${SOURCES})

# 为include/common下的文件创建子分组
file(GLOB COMMON_HEADERS "include/common/*.h")
source_group("Public Headers/Common" FILES ${COMMON_HEADERS})

# 为include/Font和Image创建子分组（如果存在）
file(GLOB FONT_HEADERS "include/Font/*.h")
file(GLOB IMAGE_HEADERS "include/Image/*.h")
if(FONT_HEADERS)
    source_group("Public Headers/Font" FILES ${FONT_HEADERS})
endif()
if(IMAGE_HEADERS)
    source_group("Public Headers/Image" FILES ${IMAGE_HEADERS})
endif()

# ============================================================================
# 创建共享库
# ============================================================================
add_library(${PROJECT_NAME} SHARED 
    ${ALL_HEADERS}
    ${SOURCES}
)

# ============================================================================
# 包含目录配置
# ============================================================================
target_include_directories(${PROJECT_NAME} 
    PUBLIC 
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/common>
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/DeveloperComponents>
        $<INSTALL_INTERFACE:include>
        $<INSTALL_INTERFACE:include/DeveloperComponents>
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/private
)

# ============================================================================
# 链接库配置
# ============================================================================
target_link_libraries(${PROJECT_NAME} 
    PUBLIC
        Qt${QT_VERSION_MAJOR}::Core
        Qt${QT_VERSION_MAJOR}::Widgets
        Qt${QT_VERSION_MAJOR}::Gui
    PRIVATE
        Qt${QT_VERSION_MAJOR}::Network
        Qt${QT_VERSION_MAJOR}::Sql
        Qt${QT_VERSION_MAJOR}::Xml
        Qt${QT_VERSION_MAJOR}::Concurrent
        Qt${QT_VERSION_MAJOR}::Multimedia
)

# Windows 特定库链接
if(WIN32)
    target_link_libraries(${PROJECT_NAME} 
        PRIVATE
            ${DWMAPI_LIB}
            ${USER32_LIB}
            ${KERNEL32_LIB}
            ${NTDLL_LIB}
    )
endif()

# ============================================================================
# 编译定义
# ============================================================================
target_compile_definitions(${PROJECT_NAME} 
    PRIVATE 
        HESWIDGETSDESIGN_LIBRARY      # 导出库符号
        QT_DEPRECATED_WARNINGS        # Qt 过时警告
        QT_DISABLE_DEPRECATED_BEFORE=0x060000  # 禁用 Qt 6.0 之前的过时 API
)

# Windows 特定定义
if(WIN32)
    target_compile_definitions(${PROJECT_NAME} 
        PRIVATE
            _WINDOWS
            _UNICODE
            UNICODE
    )
endif()

# ============================================================================
# 目标属性设置
# ============================================================================
set_target_properties(${PROJECT_NAME} PROPERTIES
    VERSION ${PROJECT_VERSION}
    SOVERSION ${PROJECT_VERSION_MAJOR}
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
    POSITION_INDEPENDENT_CODE ON
    
    # 输出目录
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib  
    ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib
    
    # Windows 特定属性
    WIN32_EXECUTABLE FALSE
    
    # 调试信息
    DEBUG_POSTFIX "d"
)

# ============================================================================
# 安装配置
# ============================================================================

# 安装库文件
install(TARGETS ${PROJECT_NAME}
    EXPORT HesWidgetsDesignTargets
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    INCLUDES DESTINATION include
)

# 安装公共头文件
install(DIRECTORY include/ 
    DESTINATION include
    FILES_MATCHING PATTERN "*.h"
    PATTERN "private" EXCLUDE
)

# 安装 DeveloperComponents 头文件
install(DIRECTORY DeveloperComponents/
    DESTINATION include/DeveloperComponents
    FILES_MATCHING PATTERN "*.h"
)

# 安装 CMake 配置文件
install(EXPORT HesWidgetsDesignTargets
    FILE HesWidgetsDesignTargets.cmake
    NAMESPACE HesWidgetsDesign::
    DESTINATION lib/cmake/HesWidgetsDesign
)

# 创建配置文件
include(CMakePackageConfigHelpers)

# 配置包配置文件
configure_package_config_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/HesWidgetsDesignConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/HesWidgetsDesignConfig.cmake"
    INSTALL_DESTINATION lib/cmake/HesWidgetsDesign
    PATH_VARS CMAKE_INSTALL_PREFIX
)

write_basic_package_version_file(
    "HesWidgetsDesignConfigVersion.cmake"
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion
)

# 安装配置文件
install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/HesWidgetsDesignConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/HesWidgetsDesignConfigVersion.cmake"
    DESTINATION lib/cmake/HesWidgetsDesign
)
